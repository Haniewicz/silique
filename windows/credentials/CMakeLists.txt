cmake_minimum_required(VERSION 3.15)
cmake_policy(SET CMP0091 NEW)
project(silique_credential_io LANGUAGES CXX)
add_library(silique_credential_io SHARED credential_io.cpp)
target_compile_features(silique_credential_io PRIVATE cxx_std_17)
target_compile_definitions(silique_credential_io PRIVATE SILIQUE_CREDENTIAL_EXPORTS)
set_property(TARGET silique_credential_io PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_executable(silique_credential_io_test credential_io_test.cpp)
target_compile_features(silique_credential_io_test PRIVATE cxx_std_17)
set_property(TARGET silique_credential_io_test PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
target_link_libraries(silique_credential_io_test PRIVATE silique_credential_io)
if(MSVC)
  target_compile_options(silique_credential_io PRIVATE /W4 /WX)
  target_compile_options(silique_credential_io_test PRIVATE /W4 /WX)
endif()
