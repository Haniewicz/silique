cmake_minimum_required(VERSION 3.15)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  cmake_policy(SET CMP0091 NEW)
  project(silique_cfa_access LANGUAGES CXX)
  enable_testing()
  set(CFA_TEST_DEFAULT ON)
else()
  set(CFA_TEST_DEFAULT OFF)
endif()

add_library(silique_cfa_core STATIC cfa_core.cpp)
target_include_directories(silique_cfa_core PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_compile_features(silique_cfa_core PUBLIC cxx_std_17)
target_link_libraries(silique_cfa_core PUBLIC advapi32)
target_compile_definitions(silique_cfa_core PUBLIC UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0A00)
function(cfa_harden target)
  set_property(TARGET ${target} PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  target_compile_options(${target} PRIVATE /W4 /WX /permissive- /EHsc /utf-8 /guard:cf)
endfunction()
cfa_harden(silique_cfa_core)

add_library(silique_cfa_windows STATIC cfa_windows.cpp)
target_link_libraries(silique_cfa_windows PUBLIC silique_cfa_core shell32 ole32 wevtapi)
cfa_harden(silique_cfa_windows)

if(NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "The CFA helper requires native x64 MSVC on Windows.")
endif()
add_executable(silique_folder_access main.cpp helper.manifest)
target_link_libraries(silique_folder_access PRIVATE silique_cfa_windows)
cfa_harden(silique_folder_access)
target_link_options(silique_folder_access PRIVATE /DYNAMICBASE /NXCOMPAT /guard:cf /CETCOMPAT /DEPENDENTLOADFLAG:0x800)

option(SILIQUE_CFA_BUILD_TESTS "Build read-only native CFA tests" ${CFA_TEST_DEFAULT})
if(SILIQUE_CFA_BUILD_TESTS)
  add_executable(silique_cfa_tests tests.cpp)
  target_link_libraries(silique_cfa_tests PRIVATE silique_cfa_windows crypt32)
  cfa_harden(silique_cfa_tests)
  target_link_options(silique_cfa_tests PRIVATE /DYNAMICBASE /NXCOMPAT /guard:cf /DEPENDENTLOADFLAG:0x800)
  add_test(NAME silique_cfa_native COMMAND silique_cfa_tests)
endif()
